Mã Nguồn ESP32 V2.0 Đồng Bộ REST API & Datasheet MQ-135 (.ino)

Đồng bộ 2 chiều toàn diện (Chế độ, Quạt, Còi, 3 LED, Mức đặt ngưỡng) với Firebase Realtime. 

#include <Arduino.h>
#include <WiFi.h>
#include <HTTPClient.h>
#include <math.h>

// ====================================================================
// HE THONG CANH BAO NONG DO CO2 TRONG HAM MO - PHIEN BAN V2 (DONG BO CLOUD)
// Tich hop cam bien MQ-135 theo chuan Datasheet & Firebase Realtime REST
// ====================================================================

// ==================================================
// 0. CAU HINH WIFI VA FIREBASE
// ==================================================
const char* WIFI_SSID = "LoVuong";
const char* WIFI_PASS = "phuancut";

// URL Realtime Database
const String FIREBASE_HOST = "https://automatic-138ef-default-rtdb.asia-southeast1.firebasedatabase.app";

// ==================================================
// 1. KHAI BAO CHAN PHAN CUNG ESP32
// ==================================================
const int MQ135_PIN = 34; // Chan Analog ADC đọc MQ-135

// L298N - Quat 1 (Thong gio vung 1)
const int IN1 = 25;
const int IN2 = 26;
const int ENA = 32;

// L298N - Quat 2 (Thong gio vung 2)
const int IN3 = 27;
const int IN4 = 14;
const int ENB = 33;

// Cum 3 Den LED Canh bao
const int LED1 = 21;
const int LED2 = 18;
const int LED3 = 23;

// Coi bao dong (Buzzer Active High)
const int BUZZER = 19;
const bool BUZZER_ACTIVE_HIGH = true;

// ==================================================
// 2. CAU HINH PWM VA HE THONG
// ==================================================
const int PWM_FREQ = 1000;
const int PWM_RES  = 10;
const int PWM_MAX  = 1023;

// Muc nguong mac dinh (User co the tuy chinh tu Web ERP qua Firebase)
int th1 = 1000; // Ngưỡng mức 1 -> mức 2
int th2 = 2000; // Ngưỡng mức 2 -> mức 3
int th3 = 3000; // Ngưỡng mức 3 -> mức 4 (Nguy hiem)

// Toc do quat 4 cap do (Dong bo chuan voi Web ERP)
int quatMuc1 = 20;  // 20%
int quatMuc2 = 50;  // 50%
int quatMuc3 = 70;  // 70%
int quatMuc4 = 100; // 100%

// ==================================================
// 3. THONG SO CHUAN DATASHEET CAM BIEN MQ-135
// ==================================================
// RL: Dien tro tai tren mach module (thuong la 10k Ohm)
const float RL_VALUE = 10.0; // kOhm
// Dien tro cam bien trong khong khi sach chuan (ADC ~ 1000 => Rs ~ 31.0 kOhm)
const float RS_CLEAN_AIR = 31.0; // kOhm

// ==================================================
// 4. BIEN TRANG THAI TOAN CUC
// ==================================================
bool isAutoMode = true;
int currentAdc = 0;
float currentVoltage = 0.0;
float currentPpm = 400.0;
int currentLevel = 1;
int currentFanSpeed = 20;

bool led1State = false;
bool led2State = false;
bool led3State = false;
bool buzzerState = false;

unsigned long lastSendTime = 0;
const unsigned long sendInterval = 1000; // Dong bo Firebase moi 1 giay

// ==================================================
// 5. HAM DIEU KHIEN PHAN CUNG
// ==================================================
void writePWM(int pin, int channel, int duty) {
#if defined(ESP_ARDUINO_VERSION) && (ESP_ARDUINO_VERSION >= ESP_ARDUINO_VERSION_VAL(3, 0, 0))
  ledcWrite(pin, duty);
#else
  ledcWrite(channel, duty);
#endif
}

void setFan(int percent) {
  currentFanSpeed = constrain(percent, 0, 100);
  int duty = map(currentFanSpeed, 0, 100, 0, PWM_MAX);

  if (currentFanSpeed <= 0) {
    digitalWrite(IN1, LOW);
    digitalWrite(IN2, LOW);
    digitalWrite(IN3, LOW);
    digitalWrite(IN4, LOW);
    writePWM(ENA, 0, 0);
    writePWM(ENB, 1, 0);
  } else {
    // Chieu quay thuan hut gio thong khi ham mo
    digitalWrite(IN1, HIGH);
    digitalWrite(IN2, LOW);
    digitalWrite(IN3, HIGH);
    digitalWrite(IN4, LOW);
    writePWM(ENA, 0, duty);
    writePWM(ENB, 1, duty);
  }
}

void applyOutputs() {
  digitalWrite(LED1, led1State ? HIGH : LOW);
  digitalWrite(LED2, led2State ? HIGH : LOW);
  digitalWrite(LED3, led3State ? HIGH : LOW);

  bool outBuzzer = BUZZER_ACTIVE_HIGH ? buzzerState : !buzzerState;
  digitalWrite(BUZZER, outBuzzer ? HIGH : LOW);
}

// Doc ADC va tinh toan trung binh loc nhieu
int readMQ135() {
  long sum = 0;
  for (int i = 0; i < 8; i++) {
    sum += analogRead(MQ135_PIN);
    delay(3);
  }
  return sum / 8;
}

// Tinh toan nong do CO2 (PPM) theo chuan Datasheet MQ-135
float calculateCO2Ppm(int adc) {
  if (adc <= 50) return 400.0;
  if (adc >= 4090) adc = 4090;

  // Dien ap A0 doc duoc tu ESP32 ADC (0 - 3.3V)
  currentVoltage = (adc / 4095.0) * 3.3;

  // Tinh dien tro cam bien Rs = RL * (4095 - ADC) / ADC
  float rs = RL_VALUE * ((4095.0 - (float)adc) / (float)adc);
  if (rs <= 0.01) rs = 0.01;

  // Ty so Rs / Rs_clean (so voi khong khi sach 400 ppm)
  float ratio = rs / RS_CLEAN_AIR;
  if (ratio <= 0.01) ratio = 0.01;

  // Cong thuc MQ-135 CO2 chuan: Baseline 400 ppm, do thi log-log doc -1.5
  float ppm = 400.0 * pow(ratio, -1.5);

  // Rang buoc gia tri thuc te trong ham mo
  if (ppm < 400.0) ppm = 400.0;
  if (ppm > 20000.0) ppm = 20000.0;

  return ppm;
}

// ==================================================
// 6. TRICH XUAT JSON AN TOAN (KHONG LOI THU VIEN NGOAI)
// ==================================================
String extractString(const String& json, const String& key) {
  int idx = json.indexOf("\"" + key + "\"");
  if (idx < 0) return "";
  int colon = json.indexOf(":", idx);
  if (colon < 0) return "";
  int q1 = json.indexOf("\"", colon);
  int q2 = json.indexOf("\"", q1 + 1);
  if (q1 >= 0 && q2 > q1) return json.substring(q1 + 1, q2);
  return "";
}

int extractInt(const String& json, const String& key, int defaultVal) {
  int idx = json.indexOf("\"" + key + "\"");
  if (idx < 0) return defaultVal;
  int colon = json.indexOf(":", idx);
  if (colon < 0) return defaultVal;
  int endComma = json.indexOf(",", colon);
  int endBrace = json.indexOf("}", colon);
  int endPos = -1;
  if (endComma > 0 && endBrace > 0) {
    endPos = (endComma < endBrace) ? endComma : endBrace;
  } else if (endComma > 0) {
    endPos = endComma;
  } else {
    endPos = endBrace;
  }
  if (endPos > colon) {
    String val = json.substring(colon + 1, endPos);
    val.trim();
    val.replace("\"", "");
    return val.toInt();
  }
  return defaultVal;
}

bool extractBool(const String& json, const String& key, bool defaultVal) {
  int idx = json.indexOf("\"" + key + "\"");
  if (idx < 0) return defaultVal;
  int colon = json.indexOf(":", idx);
  if (colon < 0) return defaultVal;
  int endComma = json.indexOf(",", colon);
  int endBrace = json.indexOf("}", colon);
  int endPos = -1;
  if (endComma > 0 && endBrace > 0) {
    endPos = (endComma < endBrace) ? endComma : endBrace;
  } else if (endComma > 0) {
    endPos = endComma;
  } else {
    endPos = endBrace;
  }
  if (endPos > colon) {
    String val = json.substring(colon + 1, endPos);
    val.trim();
    val.toLowerCase();
    val.replace("\"", "");
    if (val == "true" || val == "1") return true;
    if (val == "false" || val == "0") return false;
  }
  return defaultVal;
}

// ==================================================
// 7. DONG BO VOI FIREBASE REALTIME DATABASE (REST API)
// ==================================================
void syncFirebase() {
  if (WiFi.status() != WL_CONNECTED) return;

  HTTPClient http;

  // 1. DOC LENH DIEU KHIEN VA NGUONG TU /device_control.json (CHI 1 REQUEST CHO TAT CA)
  http.begin(FIREBASE_HOST + "/device_control.json");
  int httpCode = http.GET();
  if (httpCode == 200) {
    String payload = http.getString();

    // Mode AUTO / MANUAL
    String m = extractString(payload, "mode");
    if (m.length() > 0) {
      isAutoMode = !(m.equalsIgnoreCase("MANUAL"));
    }

    // Doc cac nguong cai dat tu nguoi dung (neu co)
    th1 = extractInt(payload, "th1", th1);
    th2 = extractInt(payload, "th2", th2);
    th3 = extractInt(payload, "th3", th3);

    // Neu o che do MANUAL: Doc day du Quat, Coi va CUM 3 LED
    if (!isAutoMode) {
      if (payload.indexOf("\"fan\"") >= 0) {
        currentFanSpeed = extractInt(payload, "fan", currentFanSpeed);
        setFan(currentFanSpeed);
      }
      if (payload.indexOf("\"buzzer\"") >= 0) {
        buzzerState = extractBool(payload, "buzzer", buzzerState);
      }
      if (payload.indexOf("\"led1\"") >= 0) {
        led1State = extractBool(payload, "led1", led1State);
      }
      if (payload.indexOf("\"led2\"") >= 0) {
        led2State = extractBool(payload, "led2", led2State);
      }
      if (payload.indexOf("\"led3\"") >= 0) {
        led3State = extractBool(payload, "led3", led3State);
      }
      applyOutputs();
    }
  }
  http.end();

  // 2. XU LY DIEU KHIEN TU DONG (KHI O CHE DO AUTO) DUA TREN CAC NGUONG DONG
  if (isAutoMode) {
    if (currentAdc < th1) {
      currentLevel = 1; // Khong khi an toan
      setFan(quatMuc1);
      led1State = led2State = led3State = false;
      buzzerState = false;
    } else if (currentAdc < th2) {
      currentLevel = 2; // Canh bao nhe
      setFan(quatMuc2);
      led1State = true;
      led2State = led3State = false;
      buzzerState = false;
    } else if (currentAdc < th3) {
      currentLevel = 3; // O nhiem / Nguy co cao
      setFan(quatMuc3);
      led1State = led2State = true;
      led3State = false;
      buzzerState = false;
    } else {
      currentLevel = 4; // Nguy hiem cuc do
      setFan(quatMuc4);
      led1State = led2State = led3State = true;
      buzzerState = true;
    }
    applyOutputs();
  }

  // 3. CAP NHAT TRANG THAI DAY DU LEN /device_status.json
  String jsonData = "{";
  jsonData += "\"mode\":\"" + String(isAutoMode ? "AUTO" : "MANUAL") + "\",";
  jsonData += "\"adc\":" + String(currentAdc) + ",";
  jsonData += "\"voltage\":" + String(currentVoltage, 2) + ",";
  jsonData += "\"co2_ppm\":" + String((int)currentPpm) + ",";
  jsonData += "\"level\":" + String(currentLevel) + ",";
  jsonData += "\"fan\":" + String(currentFanSpeed) + ",";
  jsonData += "\"led1\":" + String(led1State ? "true" : "false") + ",";
  jsonData += "\"led2\":" + String(led2State ? "true" : "false") + ",";
  jsonData += "\"led3\":" + String(led3State ? "true" : "false") + ",";
  jsonData += "\"buzzer\":" + String(buzzerState ? "true" : "false") + ",";
  jsonData += "\"th1\":" + String(th1) + ",";
  jsonData += "\"th2\":" + String(th2) + ",";
  jsonData += "\"th3\":" + String(th3);
  jsonData += "}";

  http.begin(FIREBASE_HOST + "/device_status.json");
  http.addHeader("Content-Type", "application/json");
  int patchCode = http.sendRequest("PATCH", jsonData);

  Serial.printf("[Cloud Sync] HTTP: %d | ADC: %d | CO2: %d PPM (~%.2fV) | Mức: %d | Quạt: %d%% | Mode: %s | LED: [%d,%d,%d] | Còi: %d\n",
    patchCode, currentAdc, (int)currentPpm, currentVoltage, currentLevel, currentFanSpeed,
    isAutoMode ? "AUTO" : "MANUAL", led1State, led2State, led3State, buzzerState);

  http.end();
}

// ==================================================
// SETUP
// ==================================================
void setup() {
  Serial.begin(115200);

  pinMode(MQ135_PIN, INPUT);
  pinMode(IN1, OUTPUT); pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT); pinMode(IN4, OUTPUT);
  pinMode(LED1, OUTPUT); pinMode(LED2, OUTPUT); pinMode(LED3, OUTPUT);
  pinMode(BUZZER, OUTPUT);

  analogReadResolution(12);

#if defined(ESP_ARDUINO_VERSION) && (ESP_ARDUINO_VERSION >= ESP_ARDUINO_VERSION_VAL(3, 0, 0))
  ledcAttach(ENA, PWM_FREQ, PWM_RES);
  ledcAttach(ENB, PWM_FREQ, PWM_RES);
#else
  ledcSetup(0, PWM_FREQ, PWM_RES);
  ledcAttachPin(ENA, 0);
  ledcSetup(1, PWM_FREQ, PWM_RES);
  ledcAttachPin(ENB, 1);
#endif

  // Khoi tao quat va LED
  setFan(quatMuc1);
  applyOutputs();

  // Ket noi WiFi
  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASS);
  Serial.print("Dang ket noi WiFi vao mang: ");
  Serial.println(WIFI_SSID);

  int retry = 0;
  while (WiFi.status() != WL_CONNECTED && retry < 25) {
    delay(500);
    Serial.print(".");
    retry++;
  }

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\n>> WiFi da ket noi thanh cong!");
    Serial.print(">> Dia chi IP ESP32: ");
    Serial.println(WiFi.localIP());
  } else {
    Serial.println("\n[Canh bao] Chua ket noi duoc WiFi. Tiep tuc chay Offline...");
  }
}

// ==================================================
// MAIN LOOP
// ==================================================
void loop() {
  // Doc cam bien va tinh toan nong do CO2 Datasheet
  currentAdc = readMQ135();
  currentPpm = calculateCO2Ppm(currentAdc);

  // Dong bo Firebase moi 1 giay
  if (millis() - lastSendTime >= sendInterval) {
    lastSendTime = millis();
    syncFirebase();
  }

  delay(50);
}
